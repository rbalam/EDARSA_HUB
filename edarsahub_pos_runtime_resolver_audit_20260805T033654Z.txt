   170	def resolve_pos_runtime_context(
   171	    unidad_identifier: str,
   172	    *,
   173	    expected_system_types: Optional[Sequence[str]] = None,
   174	) -> PosRuntimeContext:
   175	    """Resuelve una unidad activa por código canónico o nombre exacto."""
   176	    identifier = str(unidad_identifier or "").strip()
   177	    if not identifier:
   178	        raise PosRuntimeMappingError("Identificador de unidad requerido")
   179	    expected = _normalize_types(expected_system_types)
   180	    matches = [
   181	        context
   182	        for context in list_pos_runtime_contexts(system_types=expected or None)
   183	        if context.unidad_codigo.upper() == identifier.upper()
   184	        or context.unidad_nombre.upper() == identifier.upper()
   185	    ]
   186	    if len(matches) != 1:
   187	        raise PosRuntimeMappingError(
   188	            f"La unidad '{identifier}' no tiene una relación POS canónica única"
