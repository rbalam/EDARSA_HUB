    13	@contextmanager
    14	def sql_connection(settings: Settings):
    15	    if not settings.edarsahub_sql_host:
    16	        raise RuntimeError('EDARSAHUB_SQL_HOST no configurado')
    17	    if pymssql is None:
    18	        raise RuntimeError('pymssql no está instalado. Instala requirements.txt en el backend EDARSAHUB.')
    19	    conn = pymssql.connect(
    20	        server=settings.edarsahub_sql_host,
    21	        user=settings.edarsahub_sql_user,
    22	        password=settings.edarsahub_sql_password,
    23	        database=settings.edarsahub_sql_database,
    24	        port=settings.edarsahub_sql_port,
    25	        as_dict=True,
    26	    )
    27	    try:
    28	        yield conn
    29	        conn.commit()
    30	    except Exception:
    31	        conn.rollback()
    32	        raise
    33	    finally:
    34	        conn.close()

    15	class ParsedRows:
    16	    report_kind: str
    17	    sheet_name: str
    18	    rows: list[dict[str, Any]]
    19	    totals: dict[str, Decimal]

    22	def _clean_header(value: Any) -> str:
    23	    if value is None:
    24	        return ''
    25	    return str(value).replace('\n', ' ').strip()
    26	
    27	
    28	def _to_decimal(value: Any) -> Decimal:
    29	    if value is None or value == '':
    30	        return Decimal('0.00')
    31	    if isinstance(value, Decimal):
    32	        return value
    33	    if isinstance(value, (int, float)):
    34	        return Decimal(str(value)).quantize(Decimal('0.01'))
    35	    s = str(value).strip().replace('$', '').replace(',', '').replace('%', '')
    36	    if s in ('', '-'):
    37	        return Decimal('0.00')
    38	    try:
    39	        return Decimal(s).quantize(Decimal('0.01'))
    40	    except InvalidOperation:
    41	        return Decimal('0.00')
    42

    44	def _as_dict_rows(ws, header_row: int) -> Iterable[dict[str, Any]]:
    45	    # Algunos XLSX exportados por portales traen max_row/max_column inflado por estilos.
    46	    # Se limita a columnas con encabezado y se corta después de varias filas vacías consecutivas.
    47	    raw_headers = [_clean_header(ws.cell(row=header_row, column=col).value) for col in range(1, min(ws.max_column, 120) + 1)]
    48	    last_header_idx = 0
    49	    for i, h in enumerate(raw_headers, start=1):
    50	        if h:
    51	            last_header_idx = i
    52	    headers = raw_headers[:last_header_idx]
    53	    blank_streak = 0
    54	    for row_idx in range(header_row + 1, min(ws.max_row, 200000) + 1):
    55	        values = [ws.cell(row=row_idx, column=col).value for col in range(1, len(headers) + 1)]
    56	        if all(v is None or str(v).strip() == '' for v in values):
    57	            blank_streak += 1
    58	            if blank_streak >= 25:
    59	                break
    60	            continue
    61	        blank_streak = 0
    62	        record = {headers[i]: values[i] for i in range(len(headers)) if headers[i]}
    63	        record['HashRegistro'] = sha256_row(values)
    64	        yield record

    67	def parse_netpay_workbook(path: str | Path) -> list[ParsedRows]:
    68	    layout = detect_layout(path)
    69	    if not layout.ok:
    70	        raise ValueError('; '.join(layout.errors))
    71	
    72	    wb = load_workbook(path, read_only=True, data_only=True)
    73	    parsed: list[ParsedRows] = []
    74	
    75	    if layout.report_kind == 'DETALLE_TRANSACCIONES':
    76	        lay = layout.layouts[0]
    77	        ws = wb[TRANSACCIONES_SHEET]
    78	        rows = list(_as_dict_rows(ws, lay.header_row))
    79	        approved = [r for r in rows if str(r.get('Estatus de trx', '')).strip().lower() == 'aprobada'
    80	                    and str(r.get('Código de respuesta', '')).strip() in ('00', '0')]
    81	        totals = {
    82	            'rows': Decimal(len(rows)),
    83	            'approved_rows': Decimal(len(approved)),
    84	            'approved_monto_trx': sum((_to_decimal(r.get('Monto de trx')) for r in approved), Decimal('0.00')),
    85	            'approved_propina': sum((_to_decimal(r.get('Propina')) for r in approved), Decimal('0.00')),
    86	            'approved_venta_neta': sum((_to_decimal(r.get('Venta Neta')) for r in approved), Decimal('0.00')),
    87	        }
    88	        parsed.append(ParsedRows(layout.report_kind, TRANSACCIONES_SHEET, rows, totals))
    89	        return parsed
    90	
    91	    if layout.report_kind == 'DETALLE_DEPOSITOS_MOVIMIENTOS':
    92	        for lay in layout.layouts:
    93	            ws = wb[lay.sheet_name]
    94	            rows = list(_as_dict_rows(ws, lay.header_row))
    95	            if lay.sheet_name == DEPOSITOS_RESUMEN_SHEET:
    96	                totals = {
    97	                    'rows': Decimal(len(rows)),
    98	                    'monto_deposito': sum((_to_decimal(r.get('Monto Depósito')) for r in rows), Decimal('0.00')),
    99	                }
   100	            else:
   101	                totals = {
   102	                    'rows': Decimal(len(rows)),
   103	                    'monto_deposito': sum((_to_decimal(r.get('Monto Depósito')) for r in rows), Decimal('0.00')),
   104	                    'monto_trx': sum((_to_decimal(r.get('Monto de Trx')) for r in rows), Decimal('0.00')),
   105	                    'venta_neta': sum((_to_decimal(r.get('Venta Neta')) for r in rows), Decimal('0.00')),
   106	                    'propina': sum((_to_decimal(r.get('Propina')) for r in rows), Decimal('0.00')),
   107	                    'comision_total': sum((_to_decimal(r.get('Comisión Total ($)')) for r in rows), Decimal('0.00')),
   108	                    'iva_comisiones': sum((_to_decimal(r.get('IVA Comisiones (16%)')) for r in rows), Decimal('0.00')),
   109	                    'comisiones_mas_iva': sum((_to_decimal(r.get('Comisiones + IVA')) for r in rows), Decimal('0.00')),
   110	                }
   111	            parsed.append(ParsedRows(layout.report_kind, lay.sheet_name, rows, totals))
   112	        return parsed

    17	class NetPayRobotPipeline:
    18	    def __init__(self, settings: Settings):
    19	        self.settings = settings
    20	        self.repo = EdarsaHubRepository(settings)
    21	        self.robot = NetPayPortalRobot(settings, logger=self._print_log)
    22	
    23	    def _print_log(self, msg: str) -> None:
    24	        print(msg)
    25	
    26	    def _get_password(self) -> str:
    27	        if self.settings.netpay_password_ciphertext:
    28	            return decrypt_secret(self.settings.netpay_password_ciphertext, self.settings.server_secret_key)
    29	        if self.settings.netpay_password_plaintext:
    30	            return self.settings.netpay_password_plaintext
    31	        raise RuntimeError('No existe contraseña NetPay configurada.')
    32	
    33	    async def run(self, report_type: ReportType, date_from: date, date_to: date) -> list[DownloadResult]:
    34	        password = self._get_password()
    35	        blocks = split_date_range(date_from, date_to, max_days=31)
    36	        results: list[DownloadResult] = []
    37	        print(f'Backfill/descarga dividido en {len(blocks)} bloque(s).')
    38	        for block in blocks:
    39	            print(f'Bloque {block.number}/{block.total}: {block.date_from} - {block.date_to}')
    40	            execution_id = self.repo.create_execution(
    41	                report_type=report_type.value,
    42	                date_from=block.date_from.isoformat(),
    43	                date_to=block.date_to.isoformat(),
    44	            )
    45	            try:
    46	                result = await self.robot.run_download(report_type, block.date_from, block.date_to, password)
    47	                self.repo.register_file(execution_id, str(result.file_path), result.sha256, report_type.value)
    48	                validation = detect_layout(result.file_path)
    49	                if not validation.ok:
    50	                    self.repo.log_event(execution_id, 'ERROR', 'LAYOUT_INVALIDO', '; '.join(validation.errors))
    51	                    raise RuntimeError('; '.join(validation.errors))
    52	                parsed = parse_netpay_workbook(result.file_path)
    53	                for sheet in parsed:
    54	                    self.repo.log_event(
    55	                        execution_id,
    56	                        'INFO',
    57	                        'LAYOUT_OK',
    58	                        f'{sheet.report_kind}/{sheet.sheet_name}: {len(sheet.rows)} registros; totales={sheet.totals}',
    59	                    )
    60	                results.append(result)
    61	            except Exception as exc:
    62	                self.repo.log_event(execution_id, 'ERROR', 'BLOQUE_FALLIDO', str(exc))
    63	                raise
    64	        return results
    65	
    66	    def validate_file(self, file_path: str | Path) -> None:
    67	        layout = detect_layout(file_path)
    68	        print(layout)
    69	        if layout.ok:
    70	            parsed = parse_netpay_workbook(file_path)
    71	            for sheet in parsed:
    72	                print(f'{sheet.report_kind}/{sheet.sheet_name}')
    73	                print(f'  registros: {len(sheet.rows)}')
    74	                print(f'  totales: {sheet.totals}')
    75	        else:
    76	            raise SystemExit(2)
