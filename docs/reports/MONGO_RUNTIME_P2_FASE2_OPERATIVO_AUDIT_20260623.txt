    85	@router.post("/compras/detector/ejecutar")
    86	async def ejecutar_detector_manual(
    87	    request: EjecutarDetectorRequest = None,
    88	    credentials: HTTPAuthorizationCredentials = Depends(security)
    89	):
    90	    """
    91	    FASE 4.1 y 4.2: Ejecuta el detector de pedidos manualmente.
    92	    
    93	    Útil para pruebas y validación. Ejecuta el mismo flujo que el job automático.
    94	    
    95	    - Si se proporciona empresa_id, solo procesa esa empresa
    96	    - Si no, procesa todas las empresas activas
    97	    """
    98	    payload = verify_token(credentials.credentials)
    99	    
   100	    # Solo Administrador o Gerente puede ejecutar manualmente
   101	    user_role = payload.get("role", "")
   102	    if user_role not in ["Administrador", "Gerente", "Director"]:
   103	        raise HTTPException(
   104	            status_code=403,
   105	            detail="Solo Administrador/Gerente puede ejecutar el detector manualmente"
   106	        )
   107	    
   108	    # Obtener db async desde el scheduler manager
   109	    from core.scheduler.scheduler_manager import get_scheduler_manager
   110	    from core.scheduler.jobs.pedidos_detector_job import ejecutar_detector_manual as _ejecutar
   111	    
   112	    manager = get_scheduler_manager()
   113	    db_async = manager.db  # Conexión async de MongoDB
   114	    
   115	    empresa_id = request.empresa_id if request else None
   116	    
   117	    try:
   118	        resultado = await _ejecutar(db_async, empresa_id)
   119	        return {
   120	            "success": True,
   121	            "mensaje": "Detector ejecutado correctamente",
   122	            "estadisticas": resultado
   123	        }
   124	    except Exception as e:
   125	        raise HTTPException(status_code=500, detail=str(e))

   221	@router.post("/compras/tareas/{tarea_id}/completar")
   222	async def completar_tarea_operativa(
   223	    tarea_id: str,
   224	    credentials: HTTPAuthorizationCredentials = Depends(security)
   225	):
   226	    """
   227	    FASE 4.2: Marca una tarea operativa como completada.
   228	    
   229	    Esto debería dispararse automáticamente cuando se captura el inventario,
   230	    pero también puede hacerse manualmente.
   231	    """
   232	    payload = verify_token(credentials.credentials)
   233	    db = get_database()
   234	    
   235	    from datetime import datetime, timezone
   236	    
   237	    tarea = db.tareas_operativas_compras.find_one({"id": tarea_id})
   238	    if not tarea:
   239	        raise HTTPException(status_code=404, detail="Tarea no encontrada")
   240	    
   241	    if tarea.get("estado") == "COMPLETADA":
   242	        raise HTTPException(status_code=400, detail="Tarea ya completada")
   243	    
   244	    now = datetime.now(timezone.utc).isoformat()
   245	    
   246	    db.tareas_operativas_compras.update_one(
   247	        {"id": tarea_id},
   248	        {
   249	            "$set": {
   250	                "estado": "COMPLETADA",
   251	                "fecha_completado": now,
   252	                "completado_por": payload.get("user_id"),
   253	                "fecha_actualizacion": now
   254	            }
   255	        }
   256	    )
   257	    
   258	    return {"success": True, "mensaje": "Tarea completada"}
